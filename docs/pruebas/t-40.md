~~~java
import java.nio.file.*;
import java.util.*;
import javax.tools.*;
import javax.lang.model.element.Modifier;
import com.sun.source.tree.*;
import com.sun.source.util.*;
public class Auditoria {
 public static void main(String[] args) throws Exception {
  var compiler = ToolProvider.getSystemJavaCompiler();
  try(var manager = compiler.getStandardFileManager(null, null, java.nio.charset.StandardCharsets.UTF_8)) {
   var files = Files.walk(Path.of(args[0])).filter(p -> p.toString().endsWith(".java")).sorted().toList();
   var task = (JavacTask) compiler.getTask(null, manager, null, List.of("-proc:none"), null, manager.getJavaFileObjectsFromPaths(files));
   var units = task.parse(); var docs = DocTrees.instance(task); var positions = docs.getSourcePositions();
   int[] count = {0,0,0};
   for (var unit : units) new TreePathScanner<Void,Void>() {
    void check(Tree tree, Set<Modifier> mods) {
     boolean api = mods.contains(Modifier.PUBLIC) || mods.contains(Modifier.PROTECTED);
     if (tree instanceof MethodTree && getCurrentPath().getParentPath().getLeaf() instanceof ClassTree parent && parent.getKind() == Tree.Kind.INTERFACE) api = args.length > 1 ? mods.contains(Modifier.PUBLIC) || mods.contains(Modifier.PROTECTED) : !mods.contains(Modifier.PRIVATE);
     if (!api) return;
     var doc = docs.getDocCommentTree(getCurrentPath());
     long start = positions.getStartPosition(unit, tree);
     String name = tree instanceof ClassTree c ? c.getSimpleName().toString() : ((MethodTree)tree).getName().toString();
     String desc = doc == null ? "" : doc.getFullBody().toString().replaceAll("<[^>]*>", " ");
     int words = desc.isBlank() ? 0 : desc.strip().split("\\s+").length;
     count[0]++; if(doc == null) count[1]++; else if(words <= 3) count[2]++;
     System.out.println(Path.of(unit.getSourceFile().toUri()).getFileName()+"|"+start+"|"+name+"|"+(doc==null?"SIN":words<=3?"TRIVIAL":"OK")+"|"+desc.replace('\n',' ')+"|"+(tree instanceof MethodTree m ? m.getParameters().stream().map(v -> v.getName().toString()).collect(java.util.stream.Collectors.joining(",")) : "")+"|"+(tree instanceof MethodTree m ? String.valueOf(m.getReturnType()) : "CLASS"));
    }
    @Override public Void visitClass(ClassTree tree, Void p) { check(tree, tree.getModifiers().getFlags()); return super.visitClass(tree,p); }
    @Override public Void visitMethod(MethodTree tree, Void p) { check(tree, tree.getModifiers().getFlags()); return super.visitMethod(tree,p); }
   }.scan(unit,null);
   System.out.println("TOTAL="+count[0]+" SIN="+count[1]+" TRIVIALES="+count[2]);
  }
 }
}
~~~
