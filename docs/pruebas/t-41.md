Cada descripción comienza por `{@return …}`; el contexto complementario va después.
Se conserva el significado, la nulabilidad, unidades, restricciones y reglas documentadas.
Los retornos con contenido distinto se mantienen, incluidos `Notificacion.getUsuario`
(propiedad/autorización), `Bloqueo.getMotivo` (longitud), `NegocioException.codigo`
(clasificación del rechazo) y los booleanos de estado (resultado frente a propósito/regla).
No se interpreta como duplicación el simple hecho de describir una operación y su resultado.

## Auditoría reproducible

El auxiliar siguiente se guardó exclusivamente en TEMP como Auditoria.java;
no se incorpora ningún script a producción ni se altera la configuración de doclint.

~~~powershell
& "$env:JAVA_HOME\bin\java.exe" Auditoria.java backend/src/main/java > audit.tsv
~~~

~~~java
import java.nio.file.*;
import java.util.*;
import javax.tools.*;
import javax.lang.model.element.Modifier;
import com.sun.source.tree.*;
import com.sun.source.doctree.*;
import com.sun.source.util.*;
public class Auditoria {
 public static void main(String[] args) throws Exception {
  var compiler = ToolProvider.getSystemJavaCompiler();
  try(var manager = compiler.getStandardFileManager(null, null, java.nio.charset.StandardCharsets.UTF_8)) {
   var root = Path.of(args[0]).toAbsolutePath();
   List<Path> files;
   try(var paths = Files.walk(root)) { files = paths.filter(p -> p.toString().endsWith(".java")).sorted().toList(); }
   var task = (JavacTask) compiler.getTask(null, manager, null, List.of("-proc:none"), null, manager.getJavaFileObjectsFromPaths(files));
   var units = task.parse(); var docs = DocTrees.instance(task); var positions = docs.getSourcePositions();
   int[] count = {0,0};
   for (var unit : units) new TreePathScanner<Void,Void>() {
    void check(Tree tree, Set<Modifier> mods) {
     boolean api = mods.contains(Modifier.PUBLIC) || mods.contains(Modifier.PROTECTED);
     if (getCurrentPath().getParentPath().getLeaf() instanceof ClassTree parent && parent.getKind() == Tree.Kind.INTERFACE) api = !mods.contains(Modifier.PRIVATE);
     if (!api) return;
     var doc = docs.getDocCommentTree(getCurrentPath());
     long start = positions.getStartPosition(unit, tree);
     String name = tree instanceof ClassTree c ? c.getSimpleName().toString() : ((MethodTree)tree).getName().toString();
     String desc = doc == null ? "" : doc.getFullBody().stream().map(Object::toString).collect(java.util.stream.Collectors.joining(""));
     String ret = doc == null ? "" : doc.getBlockTags().stream().filter(t -> t instanceof com.sun.source.doctree.ReturnTree).map(t -> ((com.sun.source.doctree.ReturnTree)t).getDescription().stream().map(Object::toString).collect(java.util.stream.Collectors.joining(""))).collect(java.util.stream.Collectors.joining(""));
     long docStart = doc == null ? -1 : positions.getStartPosition(unit,doc,doc);
     count[0]++; if(doc == null) count[1]++;
     var encoder = Base64.getEncoder();
     System.out.println(root.relativize(Path.of(unit.getSourceFile().toUri())).toString().replace('\\','/')+"\t"+name+"\t"+unit.getLineMap().getLineNumber(start)+"\t"+docStart+"\t"+encoder.encodeToString(desc.getBytes(java.nio.charset.StandardCharsets.UTF_8))+"\t"+encoder.encodeToString(ret.getBytes(java.nio.charset.StandardCharsets.UTF_8))+"\t"+(doc==null?"SIN":"OK"));
    }
    @Override public Void visitClass(ClassTree tree, Void p) { check(tree, tree.getModifiers().getFlags()); return super.visitClass(tree,p); }
    @Override public Void visitMethod(MethodTree tree, Void p) { check(tree, tree.getModifiers().getFlags()); return super.visitMethod(tree,p); }
   }.scan(unit,null);
   System.out.println("TOTAL="+count[0]+" SIN="+count[1]+" FUENTES="+files.size());
  }
 }
}
~~~
